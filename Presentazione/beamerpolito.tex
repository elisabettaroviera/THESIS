    \vspace{6pt}

    \textbf{50 split} (K=5, R=10) per ogni CpG:\\[6pt]
    $\text{stability} = \dfrac{\text{split concordanti}}{\text{split validi}} \geq 0.75$\\[10pt]

    \rule{\textwidth}{0.4pt}\\[6pt]

   Score composito ($0.8$ stat $+$ $0.2$ bio):\\[4pt]
{\scriptsize $\text{score\_stat} = \tfrac{1}{2}\,\overline{\text{rank}}_p + \tfrac{1}{2}\,\overline{\text{rank}}_{|\Delta M|}$}\\[3pt]
{\scriptsize $\text{score\_bio} = \tfrac{1}{2}(1-\text{IP}) + \tfrac{1}{2}(1-\text{bio\_weight})$}\\[3pt]
{\scriptsize $\text{IP}_j = \dfrac{\Delta M(\text{Adj}-\text{Norm})_j}{\Delta M(\text{Tum}-\text{Norm})_j}$}

    Cutoff $|\Delta M|$ \textbf{data-driven} (elbow cord)\\
    $\rightarrow$ \textbf{15,659} CpG candidati

  \end{column}

\section{Dalla Feature Selection al Pannello Finale}

\begin{frame}{Selezione del Pannello: MILP Knapsack}
\begin{center}
\begin{minipage}{0.85\textwidth}
{\tiny
\begin{align*}
\max_{x} \quad & \sum_{j \in \mathcal{J}} v_j x_j + \mu\, \hat{\Phi}(x, y) \\[4pt]
\hat{\Phi}(x,y) = \quad & \frac{1}{K(2+\eta)}\!\left(\sum_{j:\,\text{prom}(j)} x_j + \sum_{j:\,\text{isl}(j)} x_j + \eta\sum_g y_g\right) \\[8pt]
\text{s.t.} \quad 
& \sum_{j} x_j = K \quad \cdot \quad x_i + x_j \leq 1 \;\forall\,(i,j):|r_{ij}|\geq 0.85 \\[4pt]
& \sum_{j \in \mathcal{J}_c} x_j \leq \delta_{\text{chr}} \;\forall c \quad \cdot \quad \sum_{j \in \mathcal{J}_A} x_j \geq K_A \\[4pt]
& x_j \leq y_{g(j)} \;\forall j \quad \cdot \quad \sum_{j:g(j)=g} x_j \leq \delta_{\max} \;\forall g \\[4pt]
& \sum_g y_g \geq G_{\min} \quad \cdot \quad \sum_{j:\Delta M_j>0} x_j \geq h_{\min} \\[4pt]
& x_j \in \{0,1\}, \quad y_g \in \{0,1\}
\end{align*}
}
\end{minipage}
\end{center}
\end{frame}

\section{Conclusioni e Sviluppi Futuri}

\begin{frame}{Conclusioni}

\begin{columns}[T]

  \begin{column}{0.48\textwidth}
    \vspace{4pt}
    \textbf{Contributo principale:}\\[6pt]
    {\small
    Framework end-to-end riproducibile:\\
    da \textbf{866k CpG} grezzi a un pannello di \textbf{50 CpG}\\
    biologicamente vincolato
    }

    \vspace{10pt}

    {\small
    \begin{itemize}
      \item[\textbf{--}] \textbf{NeuroCombat} risolve il batch effect preservando il segnale biologico
      \item[\textbf{--}] Pipeline di feature selection \textbf{stabile, non ridondante, genomicamente diversificata}
      \item[\textbf{--}] \textbf{MILP knapsack}: l'interpretabilità biologica è \textbf{imposta dai vincoli}, non emerge dalla performance
    \end{itemize}
    }

    \vspace{8pt}
    {\small
    \textbf{Il framework è generale:} i vincoli sono\\
    parametrizzabili per qualsiasi tipo tumorale
    }

  \end{column}

  \begin{column}{0.48\textwidth}
